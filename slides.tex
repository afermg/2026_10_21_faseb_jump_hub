\begin{frame}[label={sec:org0000001}]{What is Cell Painting?}
\begin{columns}[c,onlytextwidth]
\begin{column}{0.36\textwidth}
\begin{itemize}

\item Six dyes, five imaging channels.
\item Shape, intensity and texture reveal cellular responses.
\end{itemize}

\begin{frame}[label={sec:org0000003}]{Why turn images into profiles?}
Profiles are numerical descriptions of cells.

\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\begin{block}{Engineered features: CellProfiler}
\begin{itemize}
\item Size, shape, intensity, texture and spatial relationships.
\item Interpretable but somewhat redundant.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Learned representations}
\begin{itemize}
\item Features learned from images by neural networks.
\item Numerical vectors (embeddings).
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}
